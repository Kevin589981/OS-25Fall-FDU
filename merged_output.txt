   100  | void init_pid_allocator() {
   101  |     // 初始化位图，将所有位清零，表示所有PID都未被分配
   102  |     memset(pid_bitmap, 0, sizeof(pid_bitmap));
   103  |     next_pid_to_check = 1;

   107  | // 回收一个PID (此函数已是O(1)，无需修改)
   108  | void pid_recycler(int pid) {
   109  |     if (pid <= 0 || pid >= MAX_PID) {
   110  |         printk("WARNING: Attempted to recycle invalid PID %d.\n", pid);
   111  |         return;
   112  |     }
   113  | 
   114  | #ifdef PID_DEBUG
   115  |     printk("Recycling PID: %d\n", pid);
   116  | #endif
   117  | 
   118  |     int index = pid / BITS_PER_LONG;
   119  |     int offset = pid % BITS_PER_LONG;
   120  |     pid_bitmap[index] &= ~(1UL << offset);
   121  | }

   123  | //==============================================================================
   124  | //           分配一个PID (使用 AArch64 内联汇编优化)
   125  | //==============================================================================
   126  | int pid_allocator() {
   127  |     // 假设此函数总是在锁保护下调用
   128  | 
   129  |     // 从上次检查的字开始，遍历整个位图
   130  |     usize start_index = (next_pid_to_check / BITS_PER_LONG) % BITMAP_SIZE;
   131  | 
   132  |     for (usize i = 0; i < BITMAP_SIZE; ++i) {
   133  |         int index = (start_index + i) % BITMAP_SIZE;
   134  |         unsigned long word = pid_bitmap[index];
   135  | 
   136  |         // 优化: 如果这个字的所有位都已设置 (全满)，则快速跳过。
   137  |         if (word == ~0UL) {
   138  |             continue;
   139  |         }
   140  | 
   141  |         // 找到了一个不满的字，现在使用硬件指令来快速找到第一个0位。
   142  |         unsigned long inverted_word = ~word;
   143  |         unsigned long offset;
   144  | 
   145  |         // 使用AArch64内联汇编计算尾随零的数量(Count Trailing Zeros)
   146  |         // 相当于找到 inverted_word 中第一个为1的位的索引
   147  |         asm volatile (
   148  |             "rbit %0, %1\n\t"  // 1. 将位的顺序颠倒
   149  |             "clz  %0, %0\n\t"  // 2. 计算颠倒后数字的前导零数量
   150  |             : "=r" (offset)    // 输出: offset ('='表示写入, 'r'表示通用寄存器)
   151  |             : "r" (inverted_word) // 输入: inverted_word ('r'表示通用寄存器)
   152  |         );
   153  | 
   154  |         // 计算最终的PID
   155  |         int pid = index * BITS_PER_LONG + offset;
   156  | 
   157  |         // 边界检查: 确保计算出的PID没有超过最大限制
   158  |         // 这在最后一个字不是完整填充的情况下很重要
   159  |         if (pid >= MAX_PID) {
   160  |             continue; // 继续搜索下一个可能的字
   161  |         }
   162  |         
   163  |         // 找到了一个有效的、空闲的PID
   164  | 
   165  |         // 标记该位为1 (已占用)
   166  |         pid_bitmap[index] |= (1UL << offset);
   167  | 
   168  |         // 更新下一次开始搜索的位置，以实现轮转效果
   169  |         next_pid_to_check = pid + 1;
   170  |         if (next_pid_to_check >= MAX_PID) {
   171  |             next_pid_to_check = 1; // 回到开头 (跳过PID 0)
   172  |         }
   173  | 
   174  |         return pid;
   175  |     }
   176  | 
   177  |     // 如果循环一圈都没有找到空闲PID，说明PID已耗尽
   178  |     return -1; // 返回错误码
   179  | }

   288  |         bool has_children = !_empty_list(&parent->children);
   289  |         if (!has_children){
   290  |             release_spinlock(&global_process_lock);
   291  | #ifdef PID_DEBUG
   292  |             printk("194: proc.c: no children, parent pid is %d\n",parent->pid);

   294  |             return -1;

   297  |         Proc *zombie_child=NULL;
