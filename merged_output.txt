    16  |     // 注意：设备文件的类型也会是 FD_INODE。

    19  |     int ref;
    20  |     // 文件是否可读或可写。
    21  |     bool readable, writable;

    24  |         struct pipe* pipe;
    25  |         Inode* ip;
    26  |     };
    27  |     // 文件的字节偏移量。
    28  |     // 对于管道而言，该值表示已写入/读取的字节数。

    60  |  */
    61  | struct file* file_dup(struct file* f);
    62  | 
    63  | /**
    64  |     @brief 减少文件对象的引用计数。
    65  | 
    66  |     如果 f->ref == 0，则真正关闭该文件并释放索引节点（或关闭管道）。
    67  | 
    68  |     @note 由于 `cache.end_op` 可能会导致休眠，因此在调用 `end_op` 时，
    69  |     你不应该持有任何锁（指的是 `ftable` 的锁）！在释放索引节点之前，请先释放锁。
    70  | 
    71  |     @see `inode_put` 函数为索引节点（inode）执行类似的操作。
    72  |  */
    73  | void file_close(struct file* f);
    74  | 
    75  | /**
